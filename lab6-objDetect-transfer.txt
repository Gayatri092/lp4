def select_images(dataset, images_per_class):

    class_indices = {}

    for index, (_, label) in enumerate(dataset.samples):

        class_indices.setdefault(label, []).append(index)

    selected_indices = []

    for label, indices in class_indices.items():

        random.shuffle(indices)

        selected_indices.extend(
            indices[:images_per_class]
        )

    return selected_indices

def extract_features(loader):

    model.eval()

    features_list = []
    labels_list = []

    with torch.no_grad():

        for images, labels in loader:

            images = images.to(device)

            # VGG16 convolutional layers
            x = model.features(images)

            # Adaptive average pooling
            x = model.avgpool(x)

            # Flatten
            x = torch.flatten(x, 1)

            # Frozen pretrained classifier layers
            x = model.classifier[:6](x)

            features_list.append(x.cpu())
            labels_list.append(labels)

    return (
        torch.cat(features_list),
        torch.cat(labels_list)
    )

    model.classifier[6].state_dict()
)

EPOCHS = 5
PATIENCE = 2

print("Custom Classifier Training Started!")

for epoch in range(EPOCHS):

    model.classifier[6].train()

    running_loss = 0
    correct = 0
    total = 0

    for features, labels in train_feature_loader:

        features = features.to(device)
        labels = labels.to(device)

        optimizer.zero_grad()

        outputs = model.classifier[6](features)

        loss = criterion(outputs, labels)

        loss.backward()
        optimizer.step()

        running_loss += loss.item() * labels.size(0)

        predicted = outputs.argmax(dim=1)

        correct += (predicted == labels).sum().item()
        total += labels.size(0)

    train_loss = running_loss / total
    train_acc = correct / total

    # Validation
    model.classifier[6].eval()

    val_loss = 0
    val_correct = 0
    val_total = 0

    with torch.no_grad():

        for features, labels in valid_feature_loader:

            features = features.to(device)
            labels = labels.to(device)

            outputs = model.classifier[6](features)

            loss = criterion(outputs, labels)

            val_loss += loss.item() * labels.size(0)

            predicted = outputs.argmax(dim=1)

            val_correct += (predicted == labels).sum().item()
            val_total += labels.size(0)

    val_loss /= val_total
    val_acc = val_correct / val_total

    train_losses.append(train_loss)
    valid_losses.append(val_loss)

    train_accuracies.append(train_acc)
    valid_accuracies.append(val_acc)

    print(
        f"Epoch {epoch+1}/{EPOCHS} | "
        f"Train Loss: {train_loss:.4f} | "
        f"Train Accuracy: {train_acc:.4f} | "
        f"Val Loss: {val_loss:.4f} | "
        f"Val Accuracy: {val_acc:.4f}"
    )

    # Early stopping
    if val_loss < best_val_loss:

        best_val_loss = val_loss

        best_weights = copy.deepcopy(
            model.classifier[6].state_dict()
        )

        patience_counter = 0

    else:

        patience_counter += 1

    if patience_counter >= PATIENCE:

        print("Early Stopping Triggered!")
        break

# Restore best classifier weights
model.classifier[6].load_state_dict(best_weights)

print("\nClassifier Training Completed!")

def extract_features(loader):
# Function extracts deep features using pretrained VGG16.

    model.eval()
    # Switches model to evaluation mode.
    # Disables dropout behavior.

    features_list = []
    # Stores extracted features.

    labels_list = []
    # Stores corresponding labels.

    with torch.no_grad():
    # Disables gradient calculation.
    # Saves memory and computation.

        for images, labels in loader:
        # Processes batches.

            images = images.to(device)
            # Moves images to CPU or GPU.

            x = model.features(images)
            # Passes images through VGG16 convolution layers.
            # Extracts low-level and high-level visual features.

            x = model.avgpool(x)
            # Performs adaptive average pooling.

            x = torch.flatten(x, 1)
            # Converts multidimensional tensor into vector.

            x = model.classifier[:6](x)
            # Passes data through frozen classifier layers.
            # Excludes final custom classifier layer.

            features_list.append(x.cpu())
            # Stores extracted features on CPU.

            labels_list.append(labels)
            # Stores labels.

    return (
        torch.cat(features_list),
        torch.cat(labels_list)
    )
    # Combines all features and labels into single tensors.

for epoch in range(EPOCHS):
# Training loop.

    model.classifier[6].train()
    # Sets classifier to training mode.

    running_loss = 0
    correct = 0
    total = 0

    for features, labels in train_feature_loader:

        features = features.to(device)
        labels = labels.to(device)

        optimizer.zero_grad()
        # Clears old gradients.

        outputs = model.classifier[6](features)
        # Forward propagation.

        loss = criterion(outputs, labels)
        # Calculates loss.

        loss.backward()
        # Computes gradients.

        optimizer.step()
        # Updates weights.

        running_loss += loss.item() * labels.size(0)

        predicted = outputs.argmax(dim=1)
        # Finds class with highest probability.

        correct += (predicted == labels).sum().item()
        # Counts correct predictions.

        total += labels.size(0)
        # Counts processed samples.

    train_loss = running_loss / total
    # Average training loss.

    train_acc = correct / total
    # Training accuracy.

    model.classifier[6].eval()
    # Validation mode.

    val_loss = 0
    val_correct = 0
    val_total = 0

    with torch.no_grad():

        for features, labels in valid_feature_loader:

            features = features.to(device)
            labels = labels.to(device)

            outputs = model.classifier[6](features)

            loss = criterion(outputs, labels)

            val_loss += loss.item() * labels.size(0)

            predicted = outputs.argmax(dim=1)

            val_correct += (
                predicted == labels
            ).sum().item()

            val_total += labels.size(0)

    val_loss /= val_total
    # Average validation loss.

    val_acc = val_correct / val_total
    # Validation accuracy.

    train_losses.append(train_loss)
    valid_losses.append(val_loss)

    train_accuracies.append(train_acc)
    valid_accuracies.append(val_acc)

        f"Train Loss: {train_loss:.4f} | "
        f"Train Accuracy: {train_acc:.4f} | "
        f"Val Loss: {val_loss:.4f} | "
        f"Val Accuracy: {val_acc:.4f}"
    )

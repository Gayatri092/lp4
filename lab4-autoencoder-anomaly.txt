# ANOMALY DETECTION USING AUTOENCODER
# ============================================================

# AIM:
# To implement anomaly detection using an Autoencoder.
# The encoder converts input data into a latent representation,
# and the decoder reconstructs the original input.
# Reconstruction error is used to detect anomalies.


# ============================================================
# CELL 1: INSTALL REQUIRED LIBRARIES
# ============================================================
# PURPOSE:
# Install libraries required for data preprocessing,
# model building, evaluation and visualization.

%pip install numpy pandas tensorflow scikit-learn matplotlib seaborn


# ============================================================
# CELL 2: IMPORT REQUIRED LIBRARIES
# ============================================================
# PURPOSE:
# Import necessary packages.

import numpy as np
import pandas as pd
import tensorflow as tf
import matplotlib.pyplot as plt
import seaborn as sns

from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import confusion_matrix, classification_report

from tensorflow.keras import layers, models

print("TensorFlow Version:", tf.__version__)


# ============================================================
# CELL 3: LOAD ECG DATASET
# ============================================================
# PURPOSE:
# Load ECG dataset using Pandas.
#
# Place ecg.xls in the same folder as your Jupyter Notebook.
# Although the extension is .xls, the uploaded file contains
# comma-separated values, so we use read_csv().
#
# Dataset contains:
# 4998 records
# 140 ECG features
# 1 binary label column

ecg_dataset = pd.read_csv("ecg.csv", header=None)

# Display first five records
print("First Five Records:")
print(ecg_dataset.head())

# Display dataset dimensions
print("\nDataset Shape:", ecg_dataset.shape)

# Check missing values
print("\nMissing Values:", ecg_dataset.isnull().sum().sum())

# Display binary label distribution
print("\nLabel Distribution:")
print(ecg_dataset.iloc[:, -1].value_counts())


# ============================================================
# CELL 4: SEPARATE FEATURES AND LABELS
# ============================================================
# PURPOSE:
# Separate ECG features and actual class labels.
#
# First 140 columns contain ECG signal features.
# Last column contains binary class labels.
#
# IMPORTANT:
# Verify that label 0 means normal and label 1 means anomaly
# from the dataset documentation before running this code.

X = ecg_dataset.iloc[:, :-1].values
labels = ecg_dataset.iloc[:, -1].values

# Set normal class label according to dataset documentation
NORMAL_LABEL = 1

print("\nECG Features Shape:", X.shape)
print("Labels Shape:", labels.shape)


# ============================================================
# CELL 5: SPLIT TRAINING AND TESTING DATA
# ============================================================
# PURPOSE:
# Divide dataset into training and testing sets.
#
# 80% data is used for training.
# 20% data is used for testing.
#
# Stratify maintains the class distribution in both sets.

X_train_all, X_test_raw, y_train_labels, y_test_labels = \
    train_test_split(
        X,
        labels,
        test_size=0.2,
        random_state=42,
        stratify=labels
    )

# Select only normal ECG samples for Autoencoder training
X_train_normal_raw = X_train_all[
    y_train_labels == NORMAL_LABEL
]

# Divide normal training data into training and validation sets
X_train_normal_raw, X_val_normal_raw = train_test_split(
    X_train_normal_raw,
    test_size=0.1,
    random_state=42
)

print("\nNormal Training Shape:", X_train_normal_raw.shape)
print("Normal Validation Shape:", X_val_normal_raw.shape)
print("Testing Shape:", X_test_raw.shape)


# ============================================================
# CELL 6: DATA PREPROCESSING
# ============================================================
# PURPOSE:
# Standardize ECG feature values using StandardScaler.
#
# StandardScaler transforms values to approximately
# zero mean and unit standard deviation.
#
# Fit the scaler only on normal training data to prevent
# information leakage.

scaler = StandardScaler()

X_train_normal = scaler.fit_transform(X_train_normal_raw)

X_val_normal = scaler.transform(X_val_normal_raw)

X_test = scaler.transform(X_test_raw)

# Display preprocessed data shapes
print("\nPreprocessed Training Shape:", X_train_normal.shape)
print("Preprocessed Validation Shape:", X_val_normal.shape)
print("Preprocessed Testing Shape:", X_test.shape)


# ============================================================
# CELL 7: BUILD ENCODER
# ============================================================
# PURPOSE:
# Encoder compresses input data into a smaller latent
# representation.
#
# Architecture:
# Input -> 32 Neurons -> 16 Neurons -> 8 Neurons

input_dim = X_train_normal.shape[1]

encoder = models.Sequential([
    
    # Input layer accepts 140 ECG features
    layers.Input(shape=(input_dim,)),
    
    # First encoding layer
    layers.Dense(32, activation='relu'),
    
    # Second encoding layer
    layers.Dense(16, activation='relu'),
    
    # Latent representation
    layers.Dense(8, activation='relu')
    
])

print("\nEncoder Summary:")
encoder.summary()


# ============================================================
# CELL 8: BUILD DECODER
# ============================================================
# PURPOSE:
# Decoder reconstructs the original input from
# the compressed latent representation.
#
# Architecture:
# Input -> 16 Neurons -> 32 Neurons -> 140 Neurons

decoder = models.Sequential([
    
    # Input is the 8-dimensional latent representation
    layers.Input(shape=(8,)),
    
    # First decoding layer
    layers.Dense(16, activation='relu'),
    
    # Second decoding layer
    layers.Dense(32, activation='relu'),
    
    # Reconstruct original 140 features
    # Linear activation allows continuous output values
    layers.Dense(input_dim, activation='linear')
    
])

print("\nDecoder Summary:")
decoder.summary()


# ============================================================
# CELL 9: BUILD COMPLETE AUTOENCODER
# ============================================================
# PURPOSE:
# Combine encoder and decoder into one Autoencoder.
#
# Input ECG -> Encoder -> Latent Representation
# -> Decoder -> Reconstructed ECG

# Combine encoder and decoder
autoencoder = models.Sequential([
    encoder,
    decoder
])

# Explicitly build the complete model
autoencoder.build(input_shape=(None, input_dim))

# Display complete Autoencoder architecture
print("\nComplete Autoencoder Summary:")
autoencoder.summary()


# ============================================================
# CELL 10: COMPILE THE AUTOENCODER
# ============================================================
# PURPOSE:
# Configure optimizer, loss function and evaluation metric.
#
# Adam: Optimizer used to update model weights.
# MSE: Measures reconstruction error.
# MAE: Additional reconstruction evaluation metric.

autoencoder.compile(
    optimizer='adam',
    loss='mean_squared_error',
    metrics=['mae']
)


# ============================================================
# CELL 11: TRAIN THE AUTOENCODER
# ============================================================
# PURPOSE:
# Train the Autoencoder using normal ECG records.
#
# Input and output are the same because an Autoencoder
# learns to reconstruct its input.
#
# Validation data is also normal ECG data.

history = autoencoder.fit(
    X_train_normal,
    X_train_normal,
    validation_data=(X_val_normal, X_val_normal),
    epochs=20,
    batch_size=32,
    shuffle=True
)


# ============================================================
# CELL 12: CALCULATE RECONSTRUCTION ERROR
# ============================================================
# PURPOSE:
# Reconstruct testing ECG records using the trained model.
#
# MSE is calculated between original and reconstructed data.
#
# Higher reconstruction error indicates a possible anomaly.

y_pred = autoencoder.predict(X_test)

# Calculate Mean Squared Error for each testing record
mse = np.mean(np.power(X_test - y_pred, 2), axis=1)

print("\nReconstruction Errors:")
print(mse[:10])


# ============================================================
# CELL 13: DEFINE ANOMALY THRESHOLD
# ============================================================
# PURPOSE:
# Calculate threshold using normal validation data.
#
# 95th percentile is selected as the threshold.
# Records with reconstruction error above this threshold
# are classified as anomalies.

val_pred = autoencoder.predict(X_val_normal)

val_mse = np.mean(
    np.power(X_val_normal - val_pred, 2),
    axis=1
)

threshold = np.percentile(val_mse, 95)

print("\nAnomaly Threshold:", threshold)


# ============================================================
# CELL 14: DETECT ANOMALIES
# ============================================================
# PURPOSE:
# Compare testing reconstruction errors with threshold.
#
# If MSE > Threshold: Anomaly
# If MSE <= Threshold: Normal

anomalies = mse > threshold

# Count detected anomalies
num_anomalies = np.sum(anomalies)

print("\nNumber of Anomalies:", num_anomalies)

print("Number of Normal Records:", np.sum(~anomalies))


# ============================================================
# CELL 15: PLOT ANOMALY DETECTION RESULTS
# ============================================================
# PURPOSE:
# Visualize reconstruction errors and anomaly threshold.
#
# Points above the red threshold line indicate anomalies.

plt.figure(figsize=(12, 6))

plt.plot(
    mse,
    marker='o',
    linestyle='',
    markersize=3,
    label='Reconstruction Error'
)

plt.axhline(
    threshold,
    color='red',
    linestyle='--',
    label='Anomaly Threshold'
)

plt.xlabel('Sample Index')
plt.ylabel('Mean Squared Error')
plt.title('Anomaly Detection Results')

plt.legend()
plt.show()


# ============================================================
# CELL 16: DISPLAY NORMAL ECG RECONSTRUCTION
# ============================================================
# PURPOSE:
# Compare original and reconstructed normal ECG records.

normal_indices = np.where(y_test_labels == NORMAL_LABEL)[0]

if len(normal_indices) > 0:

    normal_index = normal_indices[0]

    plt.figure(figsize=(12, 6))

    plt.plot(
        X_test[normal_index],
        label='Original ECG'
    )

    plt.plot(
        y_pred[normal_index],
        label='Reconstructed ECG'
    )

    plt.xlabel('Feature Index')
    plt.ylabel('Amplitude')
    plt.title('Normal ECG Reconstruction')

    plt.legend()
    plt.show()


# ============================================================
# CELL 17: DISPLAY ANOMALOUS ECG RECONSTRUCTION
# ============================================================
# PURPOSE:
# Display original and reconstructed anomalous ECG records.
#
# Anomalous records generally have larger reconstruction
# errors than normal records.

anomalies_index = np.where(anomalies)[0]

if len(anomalies_index) > 0:

    anomaly_index = anomalies_index[0]

    plt.figure(figsize=(12, 6))

    plt.plot(
        X_test[anomaly_index],
        label='Original ECG'
    )

    plt.plot(
        y_pred[anomaly_index],
        label='Reconstructed ECG'
    )

    plt.xlabel('Feature Index')
    plt.ylabel('Amplitude')
    plt.title('Anomalous ECG Reconstruction')

    plt.legend()
    plt.show()

else:
    print("No anomalies detected in test data.")


# ============================================================
# CELL 18: EVALUATE MODEL PERFORMANCE
# ============================================================
# PURPOSE:
# Compare actual dataset labels with predicted anomaly labels.
#
# Actual anomaly = 1 if label differs from NORMAL_LABEL.
# Predicted anomaly = 1 if reconstruction error exceeds threshold.
#
# This gives a meaningful confusion matrix and classification
# report, unlike comparing predictions with themselves.

y_true = (y_test_labels != NORMAL_LABEL).astype(int)

y_predicted = anomalies.astype(int)

# Calculate confusion matrix
cm = confusion_matrix(
    y_true,
    y_predicted,
    labels=[0, 1]
)

print("\nConfusion Matrix:")
print(cm)

# Display precision, recall and F1-score
print("\nClassification Report:")

print(
    classification_report(
        y_true,
        y_predicted,
        labels=[0, 1],
        target_names=['Normal', 'Anomaly'],
        zero_division=0
    )
)


# ============================================================

# ============================================================
# PURPOSE:
# Display confusion matrix as a heatmap.

# It shows correct and incorrect normal/anomaly predictions.

plt.figure(figsize=(6, 5))

sns.heatmap(

plt.xlabel('Predicted Label')

plt.show()


# ============================================================
# CELL 20: PLOT TRAINING AND VALIDATION LOSS

plt.figure(figsize=(8, 5))

plt.plot(

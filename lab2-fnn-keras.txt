# AIM:
# To implement a Feedforward Neural Network using Keras
# and TensorFlow on the MNIST handwritten digit dataset.
# Train the model using SGD optimizer, evaluate the network
# and plot training loss and accuracy.


# ============================================================
# CELL 1: INSTALL REQUIRED LIBRARIES
# ============================================================
# PURPOSE:
# Install TensorFlow and Matplotlib.
# TensorFlow is used to build and train the neural network.
# Matplotlib is used to display images and graphs.

%pip install tensorflow matplotlib


# ============================================================
# CELL 2: IMPORT REQUIRED PACKAGES
# ============================================================
# PURPOSE:
# Import all required libraries for implementation.

import tensorflow as tf
from tensorflow import keras
import matplotlib.pyplot as plt
import numpy as np
import random

print("TensorFlow Version:", tf.__version__)


# ============================================================
# CELL 3: LOAD MNIST DATASET
# ============================================================
# PURPOSE:
# Load training and testing data from MNIST dataset.
#
# MNIST contains handwritten digit images from 0 to 9.
# Training data: 60,000 images
# Testing data: 10,000 images
# Each image has dimensions 28 x 28 pixels.

mnist = tf.keras.datasets.mnist

(x_train, y_train), (x_test, y_test) = mnist.load_data()

# Display dataset dimensions
print("Training Images:", x_train.shape)
print("Training Labels:", y_train.shape)

print("Testing Images:", x_test.shape)
print("Testing Labels:", y_test.shape)


# ============================================================
# CELL 4: DISPLAY SAMPLE IMAGE
# ============================================================
# PURPOSE:
# Display one image from the training dataset
# to understand the MNIST handwritten digit data.

plt.imshow(x_train[0], cmap='gray')
plt.title("Sample MNIST Training Image")
plt.axis('off')
plt.show()

# Display corresponding digit label
print("Actual Digit:", y_train[0])


# ============================================================
# CELL 5: DATA PREPROCESSING / NORMALIZATION
# ============================================================
# PURPOSE:
# Normalize pixel values of images.
#
# Original pixel values range from 0 to 255.
# Dividing by 255 converts values into the range 0 to 1.
#
# Normalization helps the neural network train efficiently.

x_train = x_train.astype('float32') / 255.0
x_test = x_test.astype('float32') / 255.0

# Verify normalized pixel values
print("Minimum Pixel Value:", x_train.min())
print("Maximum Pixel Value:", x_train.max())


# ============================================================
# CELL 6: DEFINE FEEDFORWARD NEURAL NETWORK ARCHITECTURE
# ============================================================
# PURPOSE:
# Create a Feedforward Neural Network using Keras Sequential API.
#
# Layer 1: Input layer accepts 28 x 28 images.
# Layer 2: Flatten converts 28 x 28 image into 784 values.
# Layer 3: Dense layer contains 128 neurons with ReLU activation.
# Layer 4: Output layer contains 10 neurons with Softmax.
#
# 10 output neurons represent digits from 0 to 9.

model = keras.Sequential([
    
    # Define input image dimensions
    keras.layers.Input(shape=(28, 28)),
    
    # Convert 2D image into 1D array
    keras.layers.Flatten(),
    
    # Hidden layer with 128 neurons and ReLU activation
    keras.layers.Dense(128, activation='relu'),
    
    # Output layer with 10 neurons for digit classification
    keras.layers.Dense(10, activation='softmax')
    
])

# Display model architecture and parameters
model.summary()


# ============================================================
# CELL 7: COMPILE THE MODEL
# ============================================================
# PURPOSE:
# Configure the model before training.
#
# Optimizer: SGD (Stochastic Gradient Descent)
# Loss: Sparse Categorical Crossentropy
# Metric: Accuracy
#
# SGD updates model weights to minimize the loss.
# Sparse categorical crossentropy is suitable for
# multi-class classification with integer labels.

model.compile(
    optimizer='sgd',
    loss='sparse_categorical_crossentropy',
    metrics=['accuracy']
)


# ============================================================
# CELL 8: TRAIN THE MODEL
# ============================================================
# PURPOSE:
# Train the neural network using training data.
#
# Epochs = 20 means the model passes through the training
# dataset 20 times.
#
# validation_split = 0.1 reserves 10% of training data
# for validation during training.
#
# The test dataset is kept separate for final evaluation.

history = model.fit(
    x_train,
    y_train,
    validation_split=0.1,
    epochs=20,
    batch_size=32
)


# ============================================================
# CELL 9: EVALUATE THE MODEL
# ============================================================
# PURPOSE:
# Evaluate the trained neural network using test data.
#
# Loss indicates prediction error.
# Accuracy indicates the percentage of correct predictions.

test_loss, test_acc = model.evaluate(x_test, y_test)

print("\nTest Results")

print("Loss = %.3f" % test_loss)

print("Accuracy = %.3f" % test_acc)


# ============================================================
# CELL 10: MAKE PREDICTIONS
# ============================================================
# PURPOSE:
# Predict handwritten digits from the testing dataset.
#
# model.predict() generates probability values for
# all 10 possible digit classes.
#
# argmax() selects the class having the highest probability.

predicted_value = model.predict(x_test, verbose=0)

# Select a random image from the testing dataset
n = random.randint(0, len(x_test) - 1)

# Find predicted digit
predicted_digit = np.argmax(predicted_value[n])

# Display selected test image
plt.imshow(x_test[n], cmap='gray')
plt.title("MNIST Test Image")
plt.axis('off')
plt.show()

# Display actual and predicted digit
print("Actual Digit:", y_test[n])
print("Predicted Digit:", predicted_digit)


# ============================================================
# CELL 11: PLOT TRAINING AND VALIDATION ACCURACY
# ============================================================
# PURPOSE:
# Plot training accuracy and validation accuracy
# across all training epochs.
#
# Accuracy indicates how correctly the model classifies digits.

plt.plot(history.history['accuracy'])

plt.plot(history.history['val_accuracy'])

plt.title('Model Accuracy')

plt.ylabel('Accuracy')

plt.xlabel('Epoch')

plt.legend(['Training Accuracy', 'Validation Accuracy'],
           loc='lower right')

plt.show()


# ============================================================
# CELL 12: PLOT TRAINING AND VALIDATION LOSS
# ============================================================
# PURPOSE:
# Plot training loss and validation loss
# across all training epochs.
#
# Loss measures the error between actual and predicted values.
# Lower loss generally indicates better model performance.

plt.plot(history.history['loss'])

plt.plot(history.history['val_loss'])

plt.title('Model Loss')

plt.ylabel('Loss')

plt.xlabel('Epoch')

plt.legend(['Training Loss', 'Validation Loss'],
           loc='upper right')

plt.show()


# ============================================================
# CELL 13: DISPLAY SAMPLE TRAINING DATA
# ============================================================
# PURPOSE:
# Display a sample training image and its corresponding label.

plt.imshow(x_train[0], cmap='gray')

plt.title("Training Image")

plt.axis('off')

plt.show()

print("Actual Training Label:", y_train[0])


# ============================================================
# CONCLUSION
# ============================================================
# Thus, we successfully implemented a Feedforward Neural
# Network using Keras and TensorFlow on the MNIST dataset.
# The model was trained using the SGD optimizer and evaluated
# using test data. Training and validation accuracy and loss
# graphs were plotted to analyze model performance.
